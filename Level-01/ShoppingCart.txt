/ Product.java
class Product {
    // Static variable for discount
    static double discount = 10.0; // Default discount in percentage

    // Final variable for product ID
    public final String productID;

    // Attributes of the Product class
    String productName;
    double price;
    int quantity;

    // Constructor to initialize the Product object
    Product(String productID, String productName, double price, int quantity) {
        this.productID = productID; // Using 'final' for unique ID
        this.productName = productName; // Using 'this' to resolve ambiguity
        this.price = price;
        this.quantity = quantity;
    }

    // Static method to update discount
    static void updateDiscount(double newDiscount) {
        if (newDiscount >= 0) {
            discount = newDiscount;
            System.out.println("Discount updated to: " + discount + "%");
        } else {
            System.out.println("Invalid discount value!");
        }
    }

    // Method to calculate the total cost for this product after discount
    double calculateTotalCost() {
        double totalCost = price * quantity;
        return totalCost - (totalCost * discount / 100); // Applying discount
    }

    // Method to display product details
    void displayProductDetails() {
        if (this instanceof Product) { // Using 'instanceof' for validation
            System.out.println("Product ID: " + productID);
            System.out.println("Product: " + productName);
            System.out.println("Price: $" + price);
            System.out.println("Quantity: " + quantity);
            System.out.println("Total Cost (after discount): $" + calculateTotalCost());
            System.out.println();
        } else {
            System.out.println("Invalid product type!");
        }
    }
}


// Cart.java
class Cart {
    // A list to hold cart items (array of CartItem objects)
    Product[] cartItems = new Product[10];
    // Limit to 10 items for simplicity
    int itemCount = 0;  // Keep track of the number of items in the cart

    // Method to add an item to the cart
    void addItemToCart(Product item) {
        if (itemCount < cartItems.length) {
            cartItems[itemCount] = item;
            itemCount++;
            System.out.println(item.ProductName + " added to cart.");
        } else {
            System.out.println("Cart is full. Cannot add more items.");
        }
    }

    // Method to remove an item from the cart by name
    void removeItemFromCart(String ProductID) {
        for (int i = 0; i < itemCount; i++) {
            if (cartItems[i].ProductID.equals(ProductID)) {
                // Shift remaining items to the left to fill the gap
                for (int j = i; j < itemCount - 1; j++) {
                    cartItems[j] = cartItems[j + 1];
                }
                cartItems[itemCount - 1] = null; // Clear the last item
                itemCount--;
                System.out.println(ProductID + " removed from cart.");
                return;
            }
        }
        System.out.println(ProductID + " not found in the cart.");
    }

    // Method to display the total cost of all items in the cart
    void displayTotalCost() {
        double totalCost = 0;
        for (int i = 0; i < itemCount; i++) {
            totalCost += cartItems[i].calculateTotalCost();
        }
        System.out.println("Total Cost of Cart: $" + totalCost);
    }
}

// ShoppingCartSystem class to test the CartItem and Cart classes
public class ShoppingCartSystem {
    public static void main(String[] args) {
         // Update the discount using the static method
        Product.updateDiscount(15.0);

        // Create Product objects
        Product product1 = new Product("P001", "Laptop", 800.00, 1);
        Product product2 = new Product("P002", "Phone", 300.00, 2);
        Product product3 = new Product("P003", "Headphones", 50.00, 3);


        // Create Cart object
        Cart cart = new Cart();

        // Add products to the cart
        cart.addItemToCart(product1);
        cart.addItemToCart(product2);
        cart.addItemToCart(product3);

        // Display details of items and total cost
        cart.displayTotalCost();

        // Remove an item from the cart
        cart.removeItemFromCart("Phone");

        // Display updated details
        cart.displayTotalCost();
    }
}
